-- 3. CREATE TRIGGER: Khởi tạo Trigger bắt sự thay đổi BOM
-- LƯU Ý: Bảng IPTH_BOM_LIST bắt buộc phải được tạo TRƯỚC KHI chạy đoạn script này.
-- Đã sử dụng cú pháp Correlated Subquery + INCLUDE_NULL_VALUES để tránh lỗi ép kiểu.
-- ==============================================================================
CREATE TRIGGER TRG_IPTH_BOM_Deep_Capture
ON IPTH_BOM_LIST
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Xử lý UPDATE
    IF EXISTS (SELECT 1 FROM inserted) AND EXISTS (SELECT 1 FROM deleted)
    BEGIN
        INSERT INTO IPTH_BOM_Change_Queue (IDBOM, Component, ChangeType, OldData, NewData)
        SELECT 
            i.IDBOM, 
            COALESCE(NULLIF(LTRIM(RTRIM(i.Component)), ''), LTRIM(RTRIM(i.BaseMaterial))),
            'UPDATE',
            (SELECT * FROM deleted d2 WHERE d2.IDBOM = i.IDBOM FOR JSON PATH, WITHOUT_ARRAY_WRAPPER, INCLUDE_NULL_VALUES),
            (SELECT * FROM inserted i2 WHERE i2.IDBOM = i.IDBOM FOR JSON PATH, WITHOUT_ARRAY_WRAPPER, INCLUDE_NULL_VALUES)
        FROM inserted i
        INNER JOIN deleted d ON i.IDBOM = d.IDBOM
        WHERE 
            (SELECT * FROM inserted i2 WHERE i2.IDBOM = i.IDBOM FOR JSON PATH, WITHOUT_ARRAY_WRAPPER, INCLUDE_NULL_VALUES) 
            <> 
            (SELECT * FROM deleted d2 WHERE d2.IDBOM = d.IDBOM FOR JSON PATH, WITHOUT_ARRAY_WRAPPER, INCLUDE_NULL_VALUES);
    END

    -- 2. Xử lý INSERT
    ELSE IF EXISTS (SELECT 1 FROM inserted) AND NOT EXISTS (SELECT 1 FROM deleted)
    BEGIN
        INSERT INTO IPTH_BOM_Change_Queue (IDBOM, Component, ChangeType, NewData)
        SELECT 
            i.IDBOM, 
            COALESCE(NULLIF(LTRIM(RTRIM(i.Component)), ''), LTRIM(RTRIM(i.BaseMaterial))),
            'INSERT', 
            (SELECT * FROM inserted i2 WHERE i2.IDBOM = i.IDBOM FOR JSON PATH, WITHOUT_ARRAY_WRAPPER, INCLUDE_NULL_VALUES)
        FROM inserted i;
    END

    -- 3. Xử lý DELETE
    ELSE IF EXISTS (SELECT 1 FROM deleted) AND NOT EXISTS (SELECT 1 FROM inserted)
    BEGIN
        INSERT INTO IPTH_BOM_Change_Queue (IDBOM, Component, ChangeType, OldData)
        SELECT 
            d.IDBOM, 
            COALESCE(NULLIF(LTRIM(RTRIM(d.Component)), ''), LTRIM(RTRIM(d.BaseMaterial))),
            'DELETE', 
            (SELECT * FROM deleted d2 WHERE d2.IDBOM = d.IDBOM FOR JSON PATH, WITHOUT_ARRAY_WRAPPER, INCLUDE_NULL_VALUES)
        FROM deleted d;
    END
END;
PRINT 'Created trigger TRG_IPTH_BOM_Deep_Capture.';
GO

/* ---------------------------------------------------------
   2. Drop UNIQUE CONSTRAINT(s) whose only key column is batch.
      This handles auto-generated constraint names safely.
   --------------------------------------------------------- */
DECLARE @ConstraintName sysname;
DECLARE @Sql nvarchar(max);

WHILE 1 = 1
BEGIN
    SET @ConstraintName = NULL;

    SELECT TOP 1
        @ConstraintName = kc.name
    FROM sys.key_constraints kc
    WHERE kc.parent_object_id = OBJECT_ID(N'dbo.wip_inventory')
      AND kc.[type] = N'UQ'
      AND
      (
          SELECT COUNT(*)
          FROM sys.index_columns ic
          WHERE ic.object_id = kc.parent_object_id
            AND ic.index_id = kc.unique_index_id
            AND ic.is_included_column = 0
      ) = 1
      AND EXISTS
      (
          SELECT 1
          FROM sys.index_columns ic
          INNER JOIN sys.columns c
              ON c.object_id = ic.object_id
             AND c.column_id = ic.column_id
          WHERE ic.object_id = kc.parent_object_id
            AND ic.index_id = kc.unique_index_id
            AND ic.is_included_column = 0
            AND c.name = N'batch'
      );

    IF @ConstraintName IS NULL
        BREAK;

    SET @Sql =
        N'ALTER TABLE dbo.wip_inventory DROP CONSTRAINT '
        + QUOTENAME(@ConstraintName)
        + N';';

    EXEC sys.sp_executesql @Sql;
END;
GO

/* ---------------------------------------------------------
   3. Drop standalone UNIQUE INDEX(es) whose only key column
      is batch. Normal idx_batch is intentionally preserved.
   --------------------------------------------------------- */
DECLARE @IndexName sysname;
DECLARE @Sql nvarchar(max);

WHILE 1 = 1
BEGIN
    SET @IndexName = NULL;

    SELECT TOP 1
        @IndexName = i.name
    FROM sys.indexes i
    WHERE i.object_id = OBJECT_ID(N'dbo.wip_inventory')
      AND i.is_unique = 1
      AND i.is_primary_key = 0
      AND i.is_unique_constraint = 0
      AND
      (
          SELECT COUNT(*)
          FROM sys.index_columns ic
          WHERE ic.object_id = i.object_id
            AND ic.index_id = i.index_id
            AND ic.is_included_column = 0
      ) = 1
      AND EXISTS
      (
          SELECT 1
          FROM sys.index_columns ic
          INNER JOIN sys.columns c
              ON c.object_id = ic.object_id
             AND c.column_id = ic.column_id
          WHERE ic.object_id = i.object_id
            AND ic.index_id = i.index_id
            AND ic.is_included_column = 0
            AND c.name = N'batch'
      );

    IF @IndexName IS NULL
        BREAK;

    SET @Sql =
        N'DROP INDEX '
        + QUOTENAME(@IndexName)
        + N' ON dbo.wip_inventory;';

    EXEC sys.sp_executesql @Sql;
END;

/*
    Notification/log cleanup timestamp setup

    Tables:
      1. dbo.DefectsLog
      2. dbo.Master_WO_Message
      3. dbo.IPTH_BOM_Change_Queue

    Policy:
      - CreatedAt is populated automatically by SQL Server.
      - UTC is used for all new timestamps.
      - Existing rows without a timestamp are assigned the deployment time,
        so they are not deleted immediately after deployment.

    Run this script once in the target MES database.
*/

SET NOCOUNT ON;
SET XACT_ABORT ON;
GO

/* ============================================================
   1. DefectsLog
   Current schema has no timestamp.
   ============================================================ */

IF OBJECT_ID(N'dbo.DefectsLog', N'U') IS NULL
BEGIN
    RAISERROR('Table dbo.DefectsLog was not found.', 16, 1);
    RETURN;
END;
GO

IF COL_LENGTH(N'dbo.DefectsLog', N'CreatedAt') IS NULL
BEGIN
    ALTER TABLE dbo.DefectsLog
    ADD CreatedAt datetime2(3) NOT NULL
        CONSTRAINT DF_DefectsLog_CreatedAt
        DEFAULT (SYSUTCDATETIME())
        WITH VALUES;
END;
GO

/* Handle the case where CreatedAt already exists but is nullable. */
UPDATE dbo.DefectsLog
SET CreatedAt = SYSUTCDATETIME()
WHERE CreatedAt IS NULL;
GO

IF EXISTS
(
    SELECT 1
    FROM sys.columns
    WHERE object_id = OBJECT_ID(N'dbo.DefectsLog')
      AND name = N'CreatedAt'
      AND is_nullable = 1
)
BEGIN
    ALTER TABLE dbo.DefectsLog
    ALTER COLUMN CreatedAt datetime2(3) NOT NULL;
END;
GO

IF NOT EXISTS
(
    SELECT 1
    FROM sys.default_constraints dc
    INNER JOIN sys.columns c
        ON c.object_id = dc.parent_object_id
       AND c.column_id = dc.parent_column_id
    WHERE dc.parent_object_id = OBJECT_ID(N'dbo.DefectsLog')
      AND c.name = N'CreatedAt'
)
BEGIN
    ALTER TABLE dbo.DefectsLog
    ADD CONSTRAINT DF_DefectsLog_CreatedAt
        DEFAULT (SYSUTCDATETIME()) FOR CreatedAt;
END;
GO

IF NOT EXISTS
(
    SELECT 1
    FROM sys.indexes
    WHERE object_id = OBJECT_ID(N'dbo.DefectsLog')
      AND name = N'IX_DefectsLog_CreatedAt'
)
BEGIN
    CREATE NONCLUSTERED INDEX IX_DefectsLog_CreatedAt
    ON dbo.DefectsLog (CreatedAt);
END;
GO

   AND dc.parent_column_id = c.column_id
